■ claps 2系をインストールする
npm install @google/clasp@2.x

■ claspにログインし権限を付与する
npx clasp login

■ .gitignore
touch .gitignore
echo node_modules >> ".gitignore"

■ Gitへの登録
# ワーキング コピーの初期化
git init

# gitアカウントの登録を行う
git config --local user.name "****"
git config --local user.email "****"

# ローカルの変更を登録する
git add .
git commit -m "Initial commit"

# branch名をmainに変更する。
git branch -M main

# リポジトリのアドレスを登録
git remote add origin "https://github.com/libraplanet/TenguCollectorGAS.git"

# リポジトリの内容を取り込み (初回結合)
git pull --rebase origin main

# push
git push -u origin main

■ ステージング前に戻す
git restore --staged .

■ clasp
# GASにpush (更新)
npx clasp push

# プロジェクトを開く
npx clasp open
